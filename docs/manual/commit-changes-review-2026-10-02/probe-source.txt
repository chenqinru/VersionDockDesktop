import { act, cleanup, fireEvent, render, screen, waitFor } from '@testing-library/react';
import { afterEach, expect, it, vi } from 'vitest';
import type { CommitNode, RepositoryStatus, WorkspaceSnapshot } from '../bindings/generated';
import { MockBridge } from '../platform/bridge';
import { useAppStore } from '../store/appStore';
import { CommitChangesWorkspace } from './CommitChangesWorkspace';
import { buildCommitFileTargets, commitKey } from '../history/commitDetails';
vi.mock('./UnifiedDiffView', () => ({ UnifiedDiffView: (props: { oldRevision?: string; newRevision?: string }) => <div data-testid="diff" data-old={props.oldRevision ?? ''} data-new={props.newRevision ?? ''} /> }));
const original = useAppStore.getState();
const repo: RepositoryStatus = { meta: { id: 'repo', name: 'Repo', rootPath: '/tmp/repo', color: '#999', kind: 'git', parentRepoId: null, depth: 0, isSubmodule: false, isWorktree: false }, branch: 'main', revision: 'a', ahead: 0, behind: 0, conflicts: 0, operation: null, files: [{ path: 'both.txt', status: 'modified', staged: true, unstaged: true, conflicted: false }] };
const snapshot: WorkspaceSnapshot = { workspace: { id: 'qa', name: 'QA', paths: ['/tmp/repo'], lastOpenedAt: '', available: true }, generation: 1, tools: { git: true, svn: true, svnadmin: true }, repositories: [repo] };
const commit = (hash: string, parents: string[] = []): CommitNode => ({ repoId: 'repo', hash, parents, shortHash: hash, message: hash, author: 'QA', email: '', authorDate: '', committerDate: '', refs: [] });
afterEach(() => { cleanup(); useAppStore.setState(original, true); });
it('working staged preview loses the INDEX revision and unstaged preview uses HEAD instead of INDEX', async () => {
 const bridge = new MockBridge((cmd) => ({ path: cmd.type === 'fileDiff' ? cmd.payload.relative_path : '', content: '@@ -1 +1 @@\n-a\n+b', language: 'text', truncated: false, binary: false, lineCount: 1 }));
 useAppStore.setState({ bridge, snapshot, changesDiffTarget: undefined, changesDiffError: undefined, changesDiffLoading: false }); useAppStore.getState().openWorkingChanges('repo');render(<CommitChangesWorkspace />);
 await screen.findByTestId('diff');expect(screen.getByTestId('diff')).toHaveAttribute('data-old','HEAD');expect(screen.getByTestId('diff')).toHaveAttribute('data-new','');
 fireEvent.click(screen.getAllByText('both.txt')[1]);await screen.findByTestId('diff');expect(screen.getByTestId('diff')).toHaveAttribute('data-old','HEAD');expect(screen.getByTestId('diff')).toHaveAttribute('data-new','');
});
it('a repeated file appears in both aggregate commit groups and both rows are highlighted', async () => {
 const commits = [commit('new', ['old']), commit('old', ['base'])]; const details=Object.fromEntries(commits.map(c=>[commitKey('repo',c.hash),{ commit:c,fullMessage:c.message,branches:{local:[],remote:[],tags:[]},files:[{path:'same.txt',status:'M',added:1,removed:1}]}]));
 const files=buildCommitFileTargets(commits,details,[repo]);expect(files).toHaveLength(1);
 useAppStore.setState({ snapshot, changes:{kind:'commits',commits,files}, changesDiff:{path:'same.txt',content:'diff',language:'text',truncated:false,binary:false,lineCount:1},changesDiffTarget:undefined,changesDiffError:undefined,changesDiffLoading:false,loadChangesDiff:vi.fn() });
 const {container}=render(<CommitChangesWorkspace />);expect(screen.getAllByText('same.txt')).toHaveLength(2);expect(container.querySelectorAll('.changes-file-row.selected')).toHaveLength(2);
});
it('Open Changes discards the active history path filter', () => {
 const c=commit('one');const detail={commit:c,fullMessage:c.message,branches:{local:[],remote:[],tags:[]},files:[{path:'filtered.txt',status:'M',added:1,removed:1},{path:'unrelated.txt',status:'M',added:1,removed:1}]};
 useAppStore.setState({snapshot,selectedCommits:[c],selectedCommitDetails:{[commitKey('repo','one')]:detail},selectedCommitLoading:{},selectedCommitError:{},historyQuery:{text:'',path:'filtered.txt'}});
 useAppStore.getState().openCommitChanges();expect(useAppStore.getState().changes?.files).toHaveLength(2);
});
it('clicking the current file while it is loading starts another request and cancels the first', async () => {
 const calls: AbortSignal[]=[];const bridge=new MockBridge((_cmd,options)=>{calls.push(options!.signal!);return new Promise(()=>{});});
 useAppStore.setState({bridge,snapshot});useAppStore.getState().openWorkingChanges('repo');render(<CommitChangesWorkspace />);await waitFor(()=>expect(calls).toHaveLength(1));
 fireEvent.click(screen.getAllByText('both.txt')[0]);await waitFor(()=>expect(calls).toHaveLength(2));expect(calls[0].aborted).toBe(true);
});
it('long file paths have no tooltip and the preview header omits the selected path', () => {
 const c=commit('one');const file={repoId:'repo',commitHash:'one',path:'very/long/path/with/an/ambiguous/file.ts',status:'M',added:1,removed:1};
 useAppStore.setState({snapshot,changes:{kind:'commits',commits:[c],files:[file]},changesDiff:undefined,changesDiffLoading:false,loadChangesDiff:vi.fn()});const {container}=render(<CommitChangesWorkspace />);
 expect(container.querySelector('.changes-file-name')).not.toHaveAttribute('title');expect(container.querySelector('.changes-workspace > header')).not.toHaveTextContent(file.path);
});
